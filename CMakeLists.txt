cmake_minimum_required(VERSION 3.21)

option(PICOLOGGER_HOST "Build portable host tests instead of firmware" OFF)
if(NOT PICOLOGGER_HOST)
    include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()

if(PICOLOGGER_HOST)
    project(picologger C)
else()
    project(picologger C CXX ASM)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(picologger_core src/core/heartbeat.c src/core/ring_buffer.c src/core/log_format.c)
target_include_directories(picologger_core PUBLIC src/core)
add_library(picologger_bme280 src/drivers/bme280.c)
target_include_directories(picologger_bme280 PUBLIC src/drivers)

if(PICOLOGGER_HOST)
    target_compile_options(picologger_core PRIVATE -Wall -Wextra -Werror)
    target_compile_options(picologger_bme280 PRIVATE -Wall -Wextra -Werror)
    include(FetchContent)
    FetchContent_Declare(unity
        GIT_REPOSITORY https://github.com/ThrowTheSwitch/Unity.git
        GIT_TAG v2.6.1
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(unity)
    enable_testing()
    add_executable(test_heartbeat test/test_heartbeat.c)
    target_compile_options(test_heartbeat PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_heartbeat PRIVATE picologger_core unity)
    add_test(NAME heartbeat COMMAND test_heartbeat)
    find_package(Threads REQUIRED)
    add_executable(test_ring_buffer test/test_ring_buffer.c)
    target_compile_options(test_ring_buffer PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_ring_buffer PRIVATE picologger_core unity Threads::Threads)
    add_test(NAME ring_buffer COMMAND test_ring_buffer)
    set_tests_properties(ring_buffer PROPERTIES TIMEOUT 120)
    add_executable(test_log_format test/test_log_format.c)
    target_compile_options(test_log_format PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_log_format PRIVATE picologger_core unity)
    add_test(NAME log_format COMMAND test_log_format)
    add_executable(test_bme280 test/test_bme280.c)
    target_compile_options(test_bme280 PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_bme280 PRIVATE picologger_bme280 unity)
    add_test(NAME bme280 COMMAND test_bme280)
    add_executable(test_i2c_bus test/test_i2c_bus.c src/hal/i2c_bus.c)
    target_include_directories(test_i2c_bus PRIVATE test/i2c_fakes src/hal)
    target_compile_options(test_i2c_bus PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_i2c_bus PRIVATE picologger_bme280 unity)
    add_test(NAME i2c_bus COMMAND test_i2c_bus)
    add_executable(test_sensor_app test/test_sensor_app.c src/app/sensor.c)
    target_include_directories(test_sensor_app PRIVATE test/i2c_fakes src/hal src/app)
    target_compile_options(test_sensor_app PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(test_sensor_app PRIVATE picologger_bme280 unity)
    add_test(NAME sensor_app COMMAND test_sensor_app)
    add_executable(generate_log_fixture test/generate_log_fixture.c)
    target_compile_options(generate_log_fixture PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(generate_log_fixture PRIVATE picologger_core)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME decode_log_python COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_SOURCE_DIR}/test/test_decode_log.py
        $<TARGET_FILE:generate_log_fixture> ${CMAKE_CURRENT_BINARY_DIR})
    foreach(board IN ITEMS pico pico_w)
        add_executable(test_board_${board} test/test_board.c src/hal/board.c)
        target_include_directories(test_board_${board} PRIVATE test/fakes src/hal)
        target_compile_options(test_board_${board} PRIVATE -Wall -Wextra -Werror)
        target_link_libraries(test_board_${board} PRIVATE picologger_core unity)
        if(board STREQUAL "pico_w")
            target_compile_definitions(test_board_${board} PRIVATE CYW43_WL_GPIO_LED_PIN=0)
        endif()
        add_test(NAME board_${board} COMMAND test_board_${board})
    endforeach()
else()
    pico_sdk_init()
    add_custom_target(picologger_version ALL
        COMMAND ${CMAKE_COMMAND} -DPROJECT_ROOT=${CMAKE_SOURCE_DIR}
                -DOUTPUT_FILE=${CMAKE_CURRENT_BINARY_DIR}/firmware_version.h
                -P ${CMAKE_SOURCE_DIR}/cmake/version.cmake
        BYPRODUCTS ${CMAKE_CURRENT_BINARY_DIR}/firmware_version.h
        VERBATIM)
    add_executable(picologger src/app/main.c src/app/sensor.c src/hal/board.c src/hal/i2c_bus.c)
    add_dependencies(picologger picologger_version)
    target_include_directories(picologger PRIVATE src/hal src/app ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(picologger PRIVATE picologger_core picologger_bme280 pico_stdlib pico_unique_id hardware_watchdog hardware_i2c)
    if(PICO_CYW43_SUPPORTED)
        target_link_libraries(picologger PRIVATE pico_cyw43_arch_none)
    endif()
    pico_enable_stdio_usb(picologger 1)
    pico_enable_stdio_uart(picologger 0)
    pico_add_extra_outputs(picologger)
endif()
